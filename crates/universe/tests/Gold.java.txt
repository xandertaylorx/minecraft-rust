import net.minecraft.stripped.universe.*;
public class Gold {
  public static void main(String[] a) {
    long seed = 12345L;
    var c = new UniverseCatalog(seed);
    long[][] ids = {{0,0,0},{1,0,0},{-3,0,5},{100,0,-42}};
    for (long[] i : ids) {
      var s = c.system(new SystemId(i[0], i[1], i[2]));
      System.out.println("S " + s.id() + " " + s.seed() + " " + s.starColor() + " " + s.bodies().size());
      for (Planet p : s.bodies())
        System.out.println(p.id().body() + " " + p.name() + " " + p.seed() + " " + p.parentBody() + " " + p.center().x + " " + p.center().y + " " + p.center().z + " " + p.radius() + " " + p.environment().temperature() + " " + p.environment().skyColor() + " " + p.environment().groundColor());
    }
  }
}
