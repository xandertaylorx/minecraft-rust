import net.minecraft.stripped.universe.*;
import net.minecraft.core.registries.BuiltInRegistries;
public class GoldTerrain {
  public static void main(String[] a) {
    net.minecraft.SharedConstants.tryDetectVersion();
    net.minecraft.server.Bootstrap.bootStrap();
    var c = new UniverseCatalog(12345L);
    long[][] ids = {{0,0,0},{1,0,0},{-3,0,5}};
    for (long[] i : ids) {
      var s = c.system(new SystemId(i[0], i[1], i[2]));
      for (Planet p : s.bodies()) {
        if (p.parentBody() == -3) continue;
        int cx = (int)Math.floor(p.nativeCenter().x), cz = (int)Math.floor(p.nativeCenter().z);
        int r = (int)p.radius();
        int[] offs = {0, 13, -29, r/2, -r/2, r-3, -(r-3), r+5};
        for (int ox : offs) for (int oz : new int[]{0, 7, -r/3, r-1}) {
          int x = cx + ox, z = cz + oz;
          int h = PlanetTerrain.height(p, x, z), sh = PlanetTerrain.surfaceHeight(p, x, z);
          StringBuilder sb = new StringBuilder("H " + p.id().system() + " " + p.id().body() + " " + x + " " + z + " " + h + " " + sh);
          int[] ys = {p.bottom()-1, p.bottom(), p.bottom()+5, p.bottom()+13, -70, -60, -50, -20, 0, 10, 30, h-9, h-5, h-3, h, h+1, 100, 230};
          for (int y : ys) {
            var st = PlanetTerrain.block(p, x, y, z, h);
            String n = BuiltInRegistries.BLOCK.getKey(st.getBlock()).getPath();
            sb.append(" ").append(y).append(":").append(n);
          }
          System.out.println(sb);
        }
      }
    }
  }
}
